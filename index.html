# GreenMap Backend • FastAPI & Orion-LD Sync
from fastapi import FastAPI, Depends, HTTPException
from app.services.orion import sync_air_quality

app = FastAPI(title="GreenMap API", version="1.0")

@app.get("/api/v1/locations/{id}/aqi")
async def get_station_aqi(id: str):
    location = await db.locations.find_one({"id": id})
    if not location:
        raise HTTPException(status_code=404)
    
    # NGSI-LD Context Broker synchronization
    aqi_data = await sync_air_quality(location)
    return {
        "status": "active",
        "aqi": aqi_data.value,
        "coords": location.coords
    }

// Task & Progress API • Laravel Form Request & Resource
namespace App\Http\Controllers\Api;

use App\Http\Requests\UpdateTaskProgressRequest;
use App\Http\Resources\TaskResource;
use App\Models\Task;

class TaskController extends Controller
{
    public function updateProgress(UpdateTaskProgressRequest $request, Task $task)
    {
        $validated = $request->validated();
        $task->update([
            'progress' => $validated['progress'],
            'status'   => $validated['progress'] == 100 ? 'DONE' : 'IN_PROGRESS'
        ]);

        return new TaskResource($task);
    }
}

LAYER 02

Dữ Liệu & Tích Hợp Hệ Thống

Mô hình hóa dữ liệu quan hệ, tích hợp các chuẩn dữ liệu mở quốc tế và cơ chế trao đổi ngữ cảnh thời gian thực.
